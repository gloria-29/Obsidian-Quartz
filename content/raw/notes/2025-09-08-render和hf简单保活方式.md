// Cloudflare Worker 脚本


// 需要保活的域名列表

const DOMAINS &#61; [

  &#39;你需要保活的域名&#39;,

];


export default {

  // 定时触发器，每天执行一次

  async scheduled(event, env, ctx) {

    console.log(&#96;[${new Date().toISOString()}] 开始执行保活任务&#96;);

    

    const results &#61; await Promise.allSettled(

      DOMAINS.map(async (domain) &#61;&gt; {

        try {

          const response &#61; await fetch(domain, {

            method: &#39;GET&#39;,

            headers: {

              &#39;User-Agent&#39;: &#39;Cloudflare-Worker-Bot&#39;,

              &#39;Cache-Control&#39;: &#39;no-cache&#39;

            },

            cf: {

              cacheTtl: 0

            }

          });


          const status &#61; response.ok ? &#39;success&#39; : &#39;failed&#39;;

          console.log(&#96;[${new Date().toISOString()}] ${domain}: ${status} (${response.status})&#96;);

          

          return {

            domain,

            status,

            statusCode: response.status,

            timestamp: new Date().toISOString()

          };

        } catch (error) {

          console.error(&#96;[${new Date().toISOString()}] ${domain}: error - ${error.message}&#96;);

          return {

            domain,

            status: &#39;error&#39;,

            error: error.message,

            timestamp: new Date().toISOString()

          };

        }

      })

    );


    console.log(&#96;[${new Date().toISOString()}] 保活任务完成&#96;);

    return results;

  },


  // HTTP请求处理，用于手动触发和查看状态

  async fetch(request, env, ctx) {

    try {

      const results &#61; await Promise.allSettled(

        DOMAINS.map(async (domain) &#61;&gt; {

          try {

            const response &#61; await fetch(domain, {

              method: &#39;GET&#39;,

              headers: {

                &#39;User-Agent&#39;: &#39;Cloudflare-Worker-Bot&#39;,

                &#39;Cache-Control&#39;: &#39;no-cache&#39;

              },

              cf: {

                cacheTtl: 0

              }

            });


            return {

              domain,

              status: response.ok ? &#39;success&#39; : &#39;failed&#39;,

              statusCode: response.status,

              timestamp: new Date().toISOString()

            };

          } catch (error) {

            return {

              domain,

              status: &#39;error&#39;,

              error: error.message,

              timestamp: new Date().toISOString()

            };

          }

        })

      );


      return new Response(JSON.stringify({

        status: &#39;completed&#39;,

        results: results.map(r &#61;&gt; r.value),

        timestamp: new Date().toISOString()

      }), {

        headers: {

          &#39;Content-Type&#39;: &#39;application/json&#39;,

          &#39;Access-Control-Allow-Origin&#39;: &#39;*&#39;

        }

      });

    } catch (error) {

      return new Response(JSON.stringify({

        status: &#39;error&#39;,

        message: error.message,

        timestamp: new Date().toISOString()

      }), {

        status: 500,

        headers: {

          &#39;Content-Type&#39;: &#39;application/json&#39;,

          &#39;Access-Control-Allow-Origin&#39;: &#39;*&#39;

        }

      });

    }

  }

};
